% ============================================================== 2 -- DRUG DISCOVERY TERMS
\begin{frame}{Drug discovery: a short vocabulary}
\begin{columns}[T,totalwidth=\textwidth]
\column{0.48\textwidth}
\sect{Ligand}
{\footnotesize A small molecule tested for interaction with a target protein.}\par\vspace{3mm}
\sect{Protein pocket}
{\footnotesize The three-dimensional region where the ligand sits and makes contacts.}\par\vspace{3mm}
\sect{Binding affinity}
{\footnotesize How strongly the ligand binds. Lower $K_d$ means tighter binding.}\par
\column{0.48\textwidth}
\sect{Relative affinity}
{\footnotesize The change between two ligands against the same target, rather than either score alone.}\par\vspace{3mm}
\sect{Congeneric / within-series}
{\footnotesize Analogues that share a chemical core and differ through small local edits.}\par\vspace{3mm}
\sect{Activity cliff}
{\footnotesize Close analogues with a large difference in measured biological activity.}\par
\end{columns}
\vfill
\noindent\colorbox{teallight}{\parbox{\dimexpr\textwidth-2\fboxsep}{\scriptsize\textbf{Our experimental label is pIC$_{50}$:} a one-unit increase means a tenfold lower IC$_{50}$ in the assay. pIC$_{50}$ is a potency measure, not a direct binding-affinity measurement.}}
\vspace{1mm}
\src{Sources: project teammate brief, pp.\ 1--3; activity-cliff definitions \parencite{stumpfe2012cliffs,maggiora2006cliffs}.}
\end{frame}

\begin{itemize}\itemsep2.2mm

% ============================================================== 8 -- THE TWO DATASETS
\begin{frame}{Two Mpro datasets, two cliff definitions}
\begin{columns}[T,totalwidth=\textwidth]
\column{0.48\textwidth}
\sect{COVID Moonshot noncovalent Mpro}
{\color{navy}\fontsize{19}{22}\selectfont\bfseries 1,102}\quad{\footnotesize molecules}\par\vspace{1.8mm}
{\footnotesize One fixed molecule split: \textbf{961 train / 141 test}. IC$_{50}$ assay values become pIC$_{50}$; the prototype uses standardized 2D structures.}\par\vspace{2mm}
\sect{Cliff rule}
{\footnotesize Morgan-fingerprint Tanimoto $\geq 0.80$ \textbf{and} measured $|\Delta\mathrm{pIC}_{50}|\geq 1$ (at least $10\times$ potency).}\par\vspace{1.5mm}
{\footnotesize \textbf{46} cliff pairs overall. Among \textbf{44} close test--test pairs, only \textbf{5} are strict held-out cliffs.}\par
\column{0.48\textwidth}
\sect{Dablander cleaned Mpro MMPs}
{\color{navy}\fontsize{19}{22}\selectfont\bfseries 1,924}\quad{\footnotesize molecules}\par\vspace{1.8mm}
{\footnotesize \textbf{12,594} supplied matched molecular pairs (MMPs). Five molecule-level folds evaluate held-out compounds; supplied pair identities are retained.}\par\vspace{2mm}
\sect{Cliff rule}
{\footnotesize Measured $|\Delta\mathrm{pIC}_{50}|\geq 2$ (at least $100\times$ potency). Noncliffs have $|\Delta|\leq 1$; intermediate pairs leave binary detection.}\par\vspace{1.5mm}
{\footnotesize \textbf{2,132} binary test--test MMPs contain \textbf{106} cliffs; \textbf{343} intermediate test--test pairs are excluded.}\par
\end{columns}
\vfill
\noindent\colorbox{teallight}{\parbox{\dimexpr\textwidth-2\fboxsep}{\scriptsize Both evaluations study Mpro activity. Their pair rules, cliff thresholds and splits differ, so the cliff rates are not pooled.}}
\vspace{1mm}
\src{Sources: Moonshot \parencite{boby2023moonshot}; Dablander MMPs \parencite{dablander2023qsar}; project protocol and result files.}
\end{frame}
